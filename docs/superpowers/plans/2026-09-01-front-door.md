# Front Door Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a single MCP front door at `POST /api/mcp` with `what_changed_since`, `coverage`, `catalogue` and `call`, put a freshness + citation envelope on every tool response, and add a script-generated changelog with a weekly gov.uk source watcher.

**Architecture:** Framework-null Vercel functions under `api/`, CommonJS, no dependencies. A new `api/_registry.js` becomes the single list of every tool across the four live servers; the per-vertical MCP handlers and the new front door both build from it. `api/_mcp.js` gains envelope injection and delegation (for `call`). Changelog and source-watching are pure modules under `api/` with thin scripts/cron wrappers so they are unit-testable without git or network.

**Tech Stack:** Node 22 (Vercel runtime; local is v24), `node --test` built-in runner (no test framework), `fetch` built-in, Vercel cron + rewrites, GitHub REST API for issues.

**Spec:** `docs/superpowers/specs/2026-09-01-front-door-design.md`

## Global Constraints

- No new npm dependencies. Tests use `node:test` and `node:assert`.
- **URLs are forever**: the four existing MCP URLs (`/api/mcp/uk-rates`, `/api/mcp/uk-exams`, `/api/mcp/provenance`, `/api/mcp/stats`) and every REST path keep working with identical tool names and inputs.
- **No fabricated data.** `data/changelog.json` starts empty; `data/sources.json` is generated by fetching live pages, never typed in.
- Front door lists **exactly 8 tools in this order**: `uk_rates_lookup, exam_topic_paper, domain_provenance, package_provenance, what_changed_since, coverage, catalogue, call`.
- Staleness thresholds: `fresh` before `nextScheduledRefresh`; `due` 0–14 days after; `overdue` beyond; live-computed tools report `live`.
- Envelope injection must never break a tool result (try/catch, fall back to bare result + attribution).
- Out-of-coverage queries return `isError:false` with `{covered:false, requested, nearest, coverage}`.
- Permalink base: `https://databutler.dev/r/<path>?<query>`; pages carry `noindex`.
- Run all tests from repo root with `node --test tests/`.
- Commit after every task. Commit messages end with the session trailer used elsewhere in this repo.

```js
'use strict';
/** Drive Vercel-style (req, res) handlers in-process. */
function safeParse(s) {
  try { return JSON.parse(s); } catch { return s; }
}

function invoke(handler, body, { method = 'POST', url = '/api/mcp', headers = {} } = {}) {
  return new Promise((resolve, reject) => {
    const req = {
      method, url, headers, body,
      on() {}, // stream API unused: body is pre-parsed like Vercel does
      [Symbol.asyncIterator]: async function* () {},
    };
    const res = {
      statusCode: 0, headers: {},
      writeHead(code, hdrs) { this.statusCode = code; this.headers = hdrs || {}; },
      end(data) {
        resolve({ status: this.statusCode, headers: this.headers, body: data === undefined ? null : safeParse(String(data)), raw: data });
      },
    };
    Promise.resolve(handler(req, res)).catch(reject);
  });
}

function rpc(handler, method, params, id = 1) {
  return invoke(handler, { jsonrpc: '2.0', id, method, ...(params ? { params } : {}) });
}

async function callTool(handler, name, args = {}) {
  const r = await rpc(handler, 'tools/call', { name, arguments: args });
  const { isError, content } = r.body.result;
  const text = content[0].text;
  return { isError, data: safeParse(text), text };
}

module.exports = { invoke, rpc, callTool };
```

```js
/**
 * Freshness block for the response envelope. Datasets declare verifiedDate
 * and nextScheduledRefresh in their JSON (see docs/CURATION.md calendar).
 */
const RATES = require('../data/uk/rates/2026-27.json');
const AQA_GCSE = require('../data/uk/exams/aqa-gcse.json');

const DUE_WINDOW_DAYS = 14;
const DATASETS = {
  'uk-rates': RATES,
  'uk-exams': AQA_GCSE,
};

function freshnessFor(dataset, now = new Date()) {
  if (dataset === 'live') {
    return { verifiedDate: now.toISOString().slice(0, 10), nextScheduledRefresh: null, staleness: 'live' };
  }
  const d = DATASETS[dataset];
  if (!d) return null;
  const { verifiedDate, nextScheduledRefresh } = d;
  const next = new Date(`${nextScheduledRefresh}T00:00:00Z`);
  const daysPast = (now - next) / 86400000;
  const staleness = daysPast < 0 ? 'fresh' : daysPast <= DUE_WINDOW_DAYS ? 'due' : 'overdue';
  return { verifiedDate, nextScheduledRefresh, staleness };
}

module.exports = { freshnessFor, DUE_WINDOW_DAYS, DATASETS };
```

```js
/**
 * Response envelope: freshness + citation permalink + attribution, added to
 * every tool result (MCP and REST). Must never break a result.
 */
const { ATTRIBUTION } = require('./_lib');
const { freshnessFor } = require('./_freshness');
const { BASE } = require('./_registry');

const SOURCE_KEYS = ['source', 'specUrl', 'officialPastPapers'];

function findSource(result) {
  if (!result || typeof result !== 'object') return null;
  for (const k of SOURCE_KEYS) if (typeof result[k] === 'string' && result[k].startsWith('http')) return result[k];
  for (const v of Object.values(result)) {
    if (Array.isArray(v)) {
      for (const item of v) { const s = findSource(item); if (s) return s; }
    } else if (v && typeof v === 'object') {
      const s = findSource(v); if (s) return s;
    }
  }
  return null;
}

function queryString(query = {}) {
  const pairs = Object.entries(query)
    .filter(([, v]) => v !== undefined && v !== null && v !== '')
    .map(([k, v]) => [k, typeof v === 'string' ? v : JSON.stringify(v)]);
  return new URLSearchParams(pairs).toString();
}

function citeFor(tool, args, freshness, result) {
  if (typeof tool.cite !== 'function') return null;
  const target = tool.cite(args || {});
  if (!target) return null;
  const qs = queryString(target.query);
  const url = `${BASE}/r/${target.path}${qs ? `?${qs}` : ''}`;
  let text = 'databutler.dev';
  if (freshness) text += freshness.staleness === 'live' ? `, computed live ${freshness.verifiedDate}` : `, verified ${freshness.verifiedDate}`;
  const source = findSource(result);
  if (source) text += ` against ${source}`;
  return { url, text };
}

function withEnvelope(tool, args, result) {
  const base = result && typeof result === 'object' && !Array.isArray(result) ? result : { result };
  try {
    const freshness = freshnessFor(tool.freshness);
    const cite = citeFor(tool, args, freshness, base);
    return { ...base, ...(freshness ? { freshness } : {}), ...(cite ? { cite } : {}), ...ATTRIBUTION };
  } catch {
    return { ...base, ...ATTRIBUTION };
  }
}

module.exports = { withEnvelope, findSource, queryString };
```

```js
/**
 * Shared stateless MCP server plumbing (streamable HTTP, JSON-RPC 2.0).
 * Same contract proven on whatbreaks.uk and lowriskquotes.com: single POST
 * per message answered with application/json; notifications get 202; no
 * sessions, no SSE. Build a handler with makeMcpHandler().
 *
 * Tool objects may carry, beyond the MCP shape: run(args), freshness, cite
 * (see _registry.js) and delegate(args) → { tool, args } — used by the front
 * door's `call` to run any registry tool while the envelope describes the
 * inner tool.
 */
const { CORS, ATTRIBUTION } = require('./_lib');
const { trackEvent } = require('./_track');
const { withEnvelope } = require('./_envelope');

const PROTOCOL_VERSION = '2025-06-18';

function makeMcpHandler({ serverName, instructions, tools }) {
  const toolDefs = tools.map(({ run, freshness, cite, delegate, ...def }) => def);
  const byName = Object.fromEntries(tools.map((t) => [t.name, t]));

  async function handle(msg) {
    const id = msg.id ?? null;
    switch (msg.method) {
      case 'initialize':
        return {
          jsonrpc: '2.0', id,
          result: {
            protocolVersion:
              typeof (msg.params || {}).protocolVersion === 'string'
                ? msg.params.protocolVersion : PROTOCOL_VERSION,
            capabilities: { tools: {} },
            serverInfo: { name: serverName, version: '1.1.0' },
            instructions,
          },
        };
      case 'ping':
        return { jsonrpc: '2.0', id, result: {} };
      case 'tools/list':
        return { jsonrpc: '2.0', id, result: { tools: toolDefs } };
      case 'tools/call': {
        const { name, arguments: args = {} } = msg.params || {};
        try {
          const def = byName[name];
          if (!def) throw new Error(`unknown tool: ${name}`);
          const target = def.delegate ? def.delegate(args) : { tool: def, args };
          if (target.tool !== def) await trackEvent(`agent_mcp_call_${String(target.tool.name).slice(0, 23)}`);
          const result = await target.tool.run(target.args);
          return {
            jsonrpc: '2.0', id,
            result: {
              content: [{ type: 'text', text: JSON.stringify(withEnvelope(target.tool, target.args, result), null, 1) }],
              isError: false,
            },
          };
        } catch (e) {
          return {
            jsonrpc: '2.0', id,
            result: { content: [{ type: 'text', text: e.message || 'tool call failed' }], isError: true },
          };
        }
      }
      default:
        return { jsonrpc: '2.0', id, error: { code: -32601, message: `method not found: ${msg.method}` } };
    }
  }

  return (req, res) => {
    if (req.method === 'OPTIONS') {
      res.writeHead(204, CORS);
      return res.end();
    }
    if (req.method !== 'POST') {
      res.writeHead(405, { 'Content-Type': 'application/json; charset=utf-8', Allow: 'POST, OPTIONS', ...CORS });
      return res.end(JSON.stringify({ info: `${serverName}: POST JSON-RPC 2.0 messages here.`, docs: ATTRIBUTION.docs }));
    }
    const respond = async (msg) => {
      if (msg.id === undefined && typeof msg.method === 'string' && msg.method.startsWith('notifications/')) {
        res.writeHead(202, CORS);
        return res.end();
      }
      if (msg.method === 'tools/call') {
        await trackEvent(`agent_mcp_${String((msg.params || {}).name).slice(0, 28)}`);
      }
      res.writeHead(200, { 'Content-Type': 'application/json; charset=utf-8', ...CORS });
      res.end(JSON.stringify(await handle(msg)));
    };
    const parseError = () => {
      res.writeHead(400, { 'Content-Type': 'application/json; charset=utf-8', ...CORS });
      res.end(JSON.stringify({ jsonrpc: '2.0', id: null, error: { code: -32700, message: 'parse error' } }));
    };
    // Vercel pre-parses req.body; fall back to the stream for local harnesses.
    if (req.body !== undefined) {
      if (typeof req.body === 'object' && req.body !== null) return respond(req.body);
      try {
        return respond(JSON.parse(String(req.body)));
      } catch {
        return parseError();
      }
    }
    let raw = '';
    req.on('data', (c) => (raw += c));
    req.on('end', () => {
      try {
        respond(JSON.parse(raw));
      } catch {
        parseError();
      }
    });
  };
}

module.exports = { makeMcpHandler, PROTOCOL_VERSION };
```

```js
/** Tiny string-similarity helpers shared by provenance and near-miss answers. */
function levenshtein(a, b) {
  const m = a.length, n = b.length;
  if (!m) return n;
  if (!n) return m;
  const dp = Array.from({ length: m + 1 }, (_, i) => [i, ...Array(n).fill(0)]);
  for (let j = 0; j <= n; j++) dp[0][j] = j;
  for (let i = 1; i <= m; i++) {
    for (let j = 1; j <= n; j++) {
      const cost = a[i - 1] === b[j - 1] ? 0 : 1;
      dp[i][j] = Math.min(dp[i - 1][j] + 1, dp[i][j - 1] + 1, dp[i - 1][j - 1] + cost);
    }
  }
  return dp[m][n];
}

/** Closest candidates to needle: substring matches first, then by edit distance. */
function nearest(needle, candidates, max = 3) {
  const q = String(needle || '').toLowerCase().trim();
  if (!q) return [];
  const scored = candidates.map((c) => {
    const s = String(c).toLowerCase();
    const sub = s.includes(q) || q.includes(s) ? 0 : 1;
    return { c, sub, d: levenshtein(q, s) };
  });
  scored.sort((x, y) => x.sub - y.sub || x.d - y.d || String(x.c).localeCompare(String(y.c)));
  return scored.slice(0, max).map((x) => x.c);
}

module.exports = { levenshtein, nearest };
```

```js
/**
 * Changelog: what changed in the datasets, and when. Entries are generated by
 * scripts/changelog.js from git diffs of data files (diffRates/diffExams) and
 * annotated by the maintainer (note). what_changed_since reads them.
 */
const CHANGELOG = require('../data/changelog.json');
const { nearest } = require('./_text');

const COVERAGE_START = '2026-08-31'; // first full verification of every dataset
const CHANGELOG_AREAS = ['uk-rates', 'uk-exams']; // provenance/stats hold no dataset
const MAX_ENTRIES = 100;

const slug = (s) => String(s).toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '');
const same = (a, b) => JSON.stringify(a ?? null) === JSON.stringify(b ?? null);
const taxYearStart = (taxYear) => `${String(taxYear).slice(0, 4)}-04-06`;

function diffRates(prev, next) {
  const out = [];
  for (const [cat, c] of Object.entries(next.categories || {})) {
    const p = prev && prev.categories ? prev.categories[cat] : null;
    const before = new Map(((p && p.figures) || []).map((f) => [slug(f.label), f]));
    for (const f of c.figures || []) {
      const was = before.get(slug(f.label));
      if (was && same(was.value, f.value) && same(was.band, f.band)) continue;
      out.push({
        area: 'uk-rates',
        key: `${cat}/${slug(f.label)}`,
        label: f.label,
        from: was ? was.value : null,
        to: f.value,
        unit: f.unit || null,
        ...(f.band ? { band: f.band } : {}),
        ...(was && !same(was.band, f.band) ? { fromBand: was.band || null } : {}),
        taxYear: next.taxYear,
        effectiveFrom: taxYearStart(next.taxYear),
        source: c.source,
        verifiedDate: next.verifiedDate,
        note: '',
      });
    }
  }
  return out;
}

function diffExams(prev, next) {
  const out = [];
  const prevSpecs = new Map(((prev && prev.specs) || []).map((s) => [s.specCode, s]));
  for (const sp of next.specs || []) {
    const base = {
      area: 'uk-exams', board: next.board, level: next.level, subject: sp.subject, specCode: sp.specCode,
      effectiveFrom: null, source: sp.specUrl, verifiedDate: next.verifiedDate, note: '',
    };
    const p = prevSpecs.get(sp.specCode);
    if (!p) {
      out.push({ ...base, key: `${sp.specCode}/spec`, label: `${next.board} ${next.level} ${sp.subject} (${sp.specCode}) added`, from: null, to: 'covered' });
      continue;
    }
    for (const paper of sp.papers || []) {
      const pp = (p.papers || []).find((x) => x.paper === paper.paper);
      const topics = (paper.topics || []).map((t) => t.name);
      const prevTopics = pp ? (pp.topics || []).map((t) => t.name) : null;
      if (!same(topics, prevTopics)) {
        out.push({ ...base, key: `${sp.specCode}/paper-${paper.paper}/topics`, label: `Paper ${paper.paper} topics`, from: prevTopics, to: topics });
      }
      for (const field of ['durationMinutes', 'marks', 'weighting', 'calculator']) {
        const a = pp ? pp[field] : undefined, b = paper[field];
        if (!same(a, b)) {
          out.push({ ...base, key: `${sp.specCode}/paper-${paper.paper}/${field}`, label: `Paper ${paper.paper} ${field}`, from: a ?? null, to: b ?? null });
        }
      }
    }
  }
  return out;
}

const entryId = (e) => `${e.area}|${e.key}|${e.verifiedDate}`;

function mergeEntries(existing, incoming) {
  const seen = new Set(existing.map(entryId));
  const added = incoming.filter((e) => !seen.has(entryId(e)));
  return { merged: [...existing, ...added], added };
}

function changesSince({ since, area = 'all' } = {}, entries = CHANGELOG) {
  const s = String(since || '').slice(0, 10);
  if (!/^\d{4}-\d{2}-\d{2}$/.test(s) || Number.isNaN(Date.parse(s))) {
    return {
      covered: false, requested: since,
      message: 'since must be an ISO date (YYYY-MM-DD) — typically your training-cutoff date',
      example: { since: '2026-06-01', area: 'all' },
    };
  }
  const a = String(area || 'all').toLowerCase().trim();
  if (a !== 'all' && !CHANGELOG_AREAS.includes(a)) {
    return { covered: false, requested: area, nearest: nearest(a, CHANGELOG_AREAS), areasCovered: CHANGELOG_AREAS };
  }
  const hits = entries
    .filter((e) => (a === 'all' || e.area === a) && (e.verifiedDate > s || (e.effectiveFrom && e.effectiveFrom > s)))
    .sort((x, y) => String(y.verifiedDate).localeCompare(String(x.verifiedDate)))
    .slice(0, MAX_ENTRIES);
  let note;
  if (s < COVERAGE_START) {
    note = `Changelog coverage starts ${COVERAGE_START} (first full verification). Changes between ${s} and ` +
      `${COVERAGE_START} are not itemised — but every figure served was verified on or after ${COVERAGE_START}, ` +
      'so call the lookup tools for current values rather than relying on memory.';
  } else if (hits.length) {
    note = 'Entries newest first: from/to, effectiveFrom, the official source, and a maintainer note where recorded.';
  } else {
    note = `No recorded changes since ${s}. Figures served are current as of their verifiedDate.`;
  }
  return { since: s, area: a, coverageStart: COVERAGE_START, areasCovered: CHANGELOG_AREAS, count: hits.length, entries: hits, note };
}

module.exports = { changesSince, diffRates, diffExams, mergeEntries, COVERAGE_START, CHANGELOG_AREAS };
```

```js
/**
 * Source-page checking shared by scripts/sources.js (writes data/sources.json)
 * and api/cron/watch-sources.js (compares against it). gov.uk pages go via
 * the JSON content API (public_updated_at + body); everything else is fetched
 * as HTML and body-hashed after stripping chrome.
 */
const crypto = require('node:crypto');

const UA = 'databutler-curation/1.0 (+https://databutler.dev)';
const TIMEOUT_MS = 8000;

function govukContentUrl(url) {
  const u = new URL(url);
  return u.hostname === 'www.gov.uk' ? `https://www.gov.uk/api/content${u.pathname}` : null;
}

function stripHtml(html) {
  return String(html)
    .replace(/<script[\s\S]*?<\/script>/gi, ' ')
    .replace(/<style[\s\S]*?<\/style>/gi, ' ')
    .replace(/<(nav|header|footer)\b[\s\S]*?<\/\1>/gi, ' ')
    .replace(/<[^>]+>/g, ' ')
    .replace(/&nbsp;/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();
}

function govukBodyText(content) {
  const d = (content && content.details) || {};
  const parts = Array.isArray(d.parts) ? d.parts.map((p) => p.body || '') : [];
  return stripHtml([d.body || '', ...parts].join(' '));
}

const hash = (s) => crypto.createHash('sha256').update(s).digest('hex').slice(0, 16);

async function fetchText(url) {
  const c = new AbortController();
  const t = setTimeout(() => c.abort(), TIMEOUT_MS);
  try {
    const r = await fetch(url, { headers: { 'User-Agent': UA, Accept: 'application/json, text/html' }, redirect: 'follow', signal: c.signal });
    if (!r.ok) throw new Error(`HTTP ${r.status}`);
    return await r.text();
  } finally {
    clearTimeout(t);
  }
}

async function checkSource(url) {
  const api = govukContentUrl(url);
  if (api) {
    try {
      const j = JSON.parse(await fetchText(api));
      return { url, updatedAt: j.public_updated_at || null, bodyHash: hash(govukBodyText(j)), via: 'govuk-content-api' };
    } catch { /* fall through to the HTML page */ }
  }
  const html = await fetchText(url);
  return { url, updatedAt: null, bodyHash: hash(stripHtml(html)), via: 'html' };
}

function sourceUrls({ rates = [], exams = [] }) {
  const urls = new Set();
  for (const r of rates) for (const c of Object.values(r.categories || {})) if (c.source) urls.add(c.source);
  for (const e of exams) for (const s of e.specs || []) if (s.specUrl) urls.add(s.specUrl);
  return [...urls];
}

function compareSource(seen, current) {
  if (!seen) return { status: 'new' };
  if ((seen.updatedAt || null) !== (current.updatedAt || null)) {
    return { status: 'changed', field: 'updatedAt', from: seen.updatedAt || null, to: current.updatedAt || null };
  }
  if (seen.bodyHash !== current.bodyHash) {
    return { status: 'changed', field: 'bodyHash', from: seen.bodyHash, to: current.bodyHash };
  }
```

```js
/**
 * Refresh data/sources.json — the "last seen" state of every source page.
 * Run after every data refresh (scripts/changelog.js calls it) so the weekly
 * watcher goes quiet again. Network required.
 *   node scripts/sources.js            # all sources

const fs = require('node:fs');
const path = require('node:path');
const { checkSource, sourceUrls } = require('../api/_sources');

const ROOT = path.resolve(__dirname, '..');
const OUT = path.join(ROOT, 'data/sources.json');

function allSourceUrls() {
  const dir = (d) => fs.readdirSync(path.join(ROOT, d)).filter((f) => f.endsWith('.json')).map((f) => JSON.parse(fs.readFileSync(path.join(ROOT, d, f), 'utf8')));
  return sourceUrls({ rates: dir('data/uk/rates'), exams: dir('data/uk/exams') });
}

async function refreshSources(urls = allSourceUrls()) {
  const checkedAt = new Date().toISOString().slice(0, 10);
  const results = await Promise.all(urls.map(async (url) => {
    try {
      return { ...(await checkSource(url)), checkedAt };
    } catch (e) {
      console.error(`  ! ${url}: ${e.message}`);
      return null;
    }
  }));
  const fresh = results.filter(Boolean);
  const existing = fs.existsSync(OUT) ? JSON.parse(fs.readFileSync(OUT, 'utf8')) : [];
  const byUrl = new Map(existing.map((e) => [e.url, e]));
  for (const e of fresh) byUrl.set(e.url, e);
  const merged = urls.map((u) => byUrl.get(u)).filter(Boolean);
  fs.writeFileSync(OUT, JSON.stringify(merged, null, 2) + '\n');
  console.log(`sources.json: ${fresh.length}/${urls.length} checked OK, ${merged.length} entries written`);
  return merged;
}
```

```js
/**
 * Generate changelog entries from the data files you just edited.
 *   1. edit data/uk/... against the live source page (docs/CURATION.md)
 *   2. node scripts/changelog.js        (add --no-fetch to skip sources.json refresh)

const { execSync } = require('node:child_process');
const fs = require('node:fs');
const path = require('node:path');
const { diffRates, diffExams, mergeEntries } = require('../api/_changelog');
const { refreshSources } = require('./sources');

const ROOT = path.resolve(__dirname, '..');
const sh = (cmd) => execSync(cmd, { cwd: ROOT, encoding: 'utf8' }).trim();
const read = (rel) => JSON.parse(fs.readFileSync(path.join(ROOT, rel), 'utf8'));

function changedDataFiles() {
  const dirty = sh('git status --porcelain').split('\n').filter(Boolean).map((l) => l.slice(3).trim());
  const outside = dirty.filter((f) => !f.startsWith('data/'));
  if (outside.length) {
    console.error('Refusing: uncommitted changes outside data/ — commit or stash them first:\n  ' + outside.join('\n  '));
    process.exit(1);
  }
  return dirty.filter((f) => f.startsWith('data/uk/') && f.endsWith('.json'));
}

/** Previous committed version; for a brand-new file, the prior sibling (e.g. last tax year). */
function previousVersion(file) {
  try {
    return JSON.parse(sh(`git show HEAD:${file}`));
  } catch { /* new file */ }
  const dir = path.dirname(file);
  const prior = fs.readdirSync(path.join(ROOT, dir))
    .filter((f) => f.endsWith('.json') && f < path.basename(file)).sort().pop();
  return prior ? read(path.join(dir, prior)) : null;
}

async function main() {
  const files = changedDataFiles();
  if (!files.length) {
    console.log('No changed data files under data/uk/. Nothing to do.');
    return;
  }
  let entries = [];
  for (const file of files) {
    const next = read(file), prev = previousVersion(file);
    const diff = file.includes('/rates/') ? diffRates(prev, next) : file.includes('/exams/') ? diffExams(prev, next) : [];
    console.log(`${file}: ${diff.length} change(s)`);
    entries = entries.concat(diff);
  }
  const clPath = path.join(ROOT, 'data/changelog.json');
  const { merged, added } = mergeEntries(read('data/changelog.json'), entries);
  fs.writeFileSync(clPath, JSON.stringify(merged, null, 2) + '\n');
  for (const e of added) console.log(`  + ${e.area} ${e.key}: ${JSON.stringify(e.from)} → ${JSON.stringify(e.to)}`);
  console.log(`\n${added.length} new entr${added.length === 1 ? 'y' : 'ies'} appended to data/changelog.json.`);
  if (added.length) console.log('Fill in each "note" (e.g. "Autumn Budget 2026"), then commit data/ together.');
  if (!process.argv.includes('--no-fetch')) await refreshSources();
}

main().catch((e) => { console.error(e); process.exit(1); });
```

```js
const esc = (s) => String(s).replace(/[&<>"']/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]));

function collectUrls(obj, out = new Set()) {
  if (!obj || typeof obj !== 'object') return out;
  for (const [k, v] of Object.entries(obj)) {
    if (typeof v === 'string' && /^https?:\/\//.test(v) && /source|url|papers|docs/i.test(k)) out.add(v);
    else if (v && typeof v === 'object') collectUrls(v, out);
  }
  return out;
}
```

```js
/** GitHub issues as the watcher's outbox. Needs GITHUB_TOKEN (issues: read/write on the repo). */
const REPO = 'physics-star-cat/databutler';
const LABEL = 'source-changed';
const ERR_TITLE = 'watcher could not check some sources';

function planIssues(results, openIssues, today = new Date().toISOString().slice(0, 10)) {
  const issues = [];
  const openText = openIssues.map((i) => `${i.title}\n${i.body || ''}`);
  for (const r of results.filter((x) => x.status === 'changed')) {
    if (openText.some((t) => t.includes(r.url))) continue;
    issues.push({
      title: `source changed: ${new URL(r.url).pathname} (${today})`,
      labels: [LABEL],
      body:
        `**URL:** ${r.url}\n\n**${r.field}:** \`${r.from}\` → \`${r.to}\`\n\n` +
        'A figure may have moved. Open the page, update the data file against it, run ' +
        '`node scripts/changelog.js`, fill in the note, commit data/ (this refreshes sources.json and closes the loop). ' +
        'If nothing material changed, just run `node scripts/sources.js` and commit.',
    });
  }
  const errs = results.filter((x) => x.status === 'error');
  if (errs.length && !openIssues.some((i) => i.title === ERR_TITLE)) {
    issues.push({ title: ERR_TITLE, labels: [LABEL], body: errs.map((e) => `- ${e.url} — ${e.error}`).join('\n') });
  }
  return issues;
}

async function gh(path, opts = {}) {
  const token = process.env.GITHUB_TOKEN;
  if (!token) throw new Error('GITHUB_TOKEN not set');
  const r = await fetch(`https://api.github.com/repos/${REPO}${path}`, {
    ...opts,
    headers: {
      Authorization: `Bearer ${token}`, Accept: 'application/vnd.github+json',
      'User-Agent': 'databutler-watcher', 'Content-Type': 'application/json', ...(opts.headers || {}),
    },
  });
  if (!r.ok) throw new Error(`GitHub ${r.status} on ${path}`);
  return r.json();
}

const listOpenIssues = () => gh(`/issues?state=open&labels=${LABEL}&per_page=100`);
```

```js
/**
 * Weekly: has any source page changed since data/sources.json was written?
 * Never edits data — files a GitHub issue per changed URL (deduped) and one
 * combined issue for URLs it could not check. ?dry=1 reports without filing.
 * Vercel cron sends Authorization: Bearer $CRON_SECRET.
 */
const { sendJson, RATES } = require('../_lib');
const AQA_GCSE = require('../../data/uk/exams/aqa-gcse.json');
const SEEN = require('../../data/sources.json');
const { checkSource, sourceUrls, compareSource } = require('../_sources');
const { planIssues, listOpenIssues, openIssue } = require('../_github');

module.exports = async (req, res) => {
  const secret = process.env.CRON_SECRET;
  const auth = (req.headers && (req.headers.authorization || req.headers.Authorization)) || '';
  if (!secret || auth !== `Bearer ${secret}`) return sendJson(res, 401, { error: 'unauthorised' });

  const dry = new URL(req.url, 'https://databutler.dev').searchParams.get('dry') === '1';
  const urls = sourceUrls({ rates: [RATES], exams: [AQA_GCSE] });
  const seenByUrl = Object.fromEntries(SEEN.map((s) => [s.url, s]));

  const results = await Promise.all(urls.map(async (url) => {
    try {
      const current = await checkSource(url);
      return { url, ...compareSource(seenByUrl[url], current) };
    } catch (e) {
      return { url, status: 'error', error: e.message };
    }
  }));

  let issues = [];
  let filing = null;
  try {
    const open = dry ? [] : await listOpenIssues();
    issues = planIssues(results, open);
    if (!dry) for (const issue of issues) await openIssue(issue);
  } catch (e) {
    filing = e.message; // GitHub down / token missing: still report what we saw
  }

  sendJson(res, 200, {
    dry,
    checked: urls.length,
    changed: results.filter((r) => r.status === 'changed'),
    unseen: results.filter((r) => r.status === 'new').map((r) => r.url),
    errors: results.filter((r) => r.status === 'error'),
    issues: issues.map((i) => i.title),
    ...(filing ? { filingError: filing } : {}),
  });
};
```

```js
/**
 * REST entry for the stats toolkit. One endpoint, ?op= selects the tool;
 * complex inputs (arrays, tables) come via POST JSON body.
 */
const { sendJson, handleOptions, queryParams, ATTRIBUTION } = require('./_lib');
const { ALL_TOOLS } = require('./_registry');
const { withEnvelope } = require('./_envelope');
const { trackEvent } = require('./_track');

const OPS = {
  descriptive: 'descriptive_stats',
  distribution: 'distribution',
  'hypothesis-test': 'hypothesis_test',
  'confidence-interval': 'confidence_interval',
  'bayes-update': 'bayes_update',
  regression: 'linear_regression',
};

module.exports = async (req, res) => {
  if (handleOptions(req, res)) return;
  await trackEvent('agent_api_stats');
  let args = queryParams(req);
  for (const k of ['params', 'data', 'expected', 'observed', 'table', 'x', 'y', 'hypotheses', 'data1', 'data2']) {
    if (typeof args[k] === 'string') { try { args[k] = JSON.parse(args[k]); } catch { /* leave as string */ } }
  }
  for (const k of ['at', 'p', 'mu0', 'successes', 'n', 'p0', 'successes1', 'n1', 'successes2', 'n2', 'alpha', 'confidence', 'mean', 'sd']) {
    if (typeof args[k] === 'string' && args[k] !== '' && !Number.isNaN(Number(args[k]))) args[k] = Number(args[k]);
  }
  if (req.method === 'POST') {
    try {
      const chunks = [];
      for await (const c of req) chunks.push(c);
      if (chunks.length) args = { ...args, ...JSON.parse(Buffer.concat(chunks).toString()) };
    } catch { /* ignore body parse */ }
  }
  const op = String(args.op || '').toLowerCase();
  const tool = ALL_TOOLS[OPS[op]];
  if (!tool) {
    return sendJson(res, 400, {
      error: `pass ?op= one of: ${Object.keys(OPS).join(', ')}. See /api for usage.`,
      ...ATTRIBUTION,
    });
  }
  const { op: _op, ...toolArgs } = args;
  try {
    sendJson(res, 200, { op, ...withEnvelope(tool, toolArgs, tool.run(toolArgs)) });
  } catch (e) {
    sendJson(res, 400, { op, error: e.message, ...ATTRIBUTION });
  }
};
```

```bash
#!/usr/bin/env bash
# Live verification of the front door. Usage: scripts/verify-hub.sh [https://databutler.dev]
set -uo pipefail
BASE="${1:-https://databutler.dev}"
MCP="$BASE/api/mcp"
FAILED=0

rpc() { curl -sS -X POST "$MCP" -H 'Content-Type: application/json' -d "$1"; }
tool() { rpc "{\"jsonrpc\":\"2.0\",\"id\":3,\"method\":\"tools/call\",\"params\":{\"name\":\"$1\",\"arguments\":$2}}"; }
inner() { python3 -c "import json,sys; d=json.loads(sys.argv[1]); print(d['result']['content'][0]['text'])" "$1"; }
check() { # label, json, python-expression over d
  if python3 -c "import json,sys; d=json.loads(sys.argv[1]); assert ($3), d" "$2" 2>/dev/null; then echo "PASS $1"; else echo "FAIL $1"; echo "$2" | head -c 500; echo; FAILED=1; fi
}

check initialize "$(rpc '{"jsonrpc":"2.0","id":1,"method":"initialize","params":{"protocolVersion":"2025-06-18","capabilities":{},"clientInfo":{"name":"verify","version":"0"}}}')" \
  "d['result']['serverInfo']['name']=='databutler' and 'what_changed_since' in d['result']['instructions']"
check tools/list "$(rpc '{"jsonrpc":"2.0","id":2,"method":"tools/list"}')" \
  "[t['name'] for t in d['result']['tools']]==['uk_rates_lookup','exam_topic_paper','domain_provenance','package_provenance','what_changed_since','coverage','catalogue','call']"
check rates "$(inner "$(tool uk_rates_lookup '{"category":"income-tax"}')")" \
  "d['freshness']['staleness'] in ('fresh','due','overdue') and d['cite']['url']=='https://databutler.dev/r/uk/rates?category=income-tax'"
check topic "$(inner "$(tool exam_topic_paper '{"topic":"electricity","board":"aqa","subject":"physics"}')")" \
  "d['results'][0]['paper']==1 and d['cite']['url'].startswith('https://databutler.dev/r/uk/exams/topic?')"
check domain "$(inner "$(tool domain_provenance '{"domain":"databutler.dev"}')")" \
  "d['freshness']['staleness']=='live' and 'cite' in d"
check package "$(inner "$(tool package_provenance '{"ecosystem":"npm","name":"express"}')")" \
  "d['freshness']['staleness']=='live'"
check changed "$(inner "$(tool what_changed_since '{"since":"2026-06-01"}')")" \
  "d['since']=='2026-06-01' and isinstance(d['entries'],list) and d['coverageStart']=='2026-08-31'"
check coverage "$(inner "$(tool coverage '{"area":"uk-exams"}')")" "'aqa' in d['boards']"
check catalogue "$(inner "$(tool catalogue '{}')")" "d['count']==12"
check call-stats "$(inner "$(tool call '{"tool":"distribution","arguments":{"dist":"normal","at":1.96}}')")" \
  "abs(d['cdf']-0.975)<0.001 and d['cite']['url'].startswith('https://databutler.dev/r/stats?op=distribution')"
check call-unknown "$(inner "$(tool call '{"tool":"regresion","arguments":{}}')")" "d['covered']==False and d['nearest'][0]=='linear_regression'"
check call-nearmiss-rates "$(inner "$(tool uk_rates_lookup '{"category":"incme"}')")" "d['covered']==False and 'income-tax' in d['nearest']"

for name in uk-rates uk-exams provenance stats; do
  check "legacy $name tools/list" "$(curl -sS -X POST "$BASE/api/mcp/$name" -H 'Content-Type: application/json' -d '{"jsonrpc":"2.0","id":2,"method":"tools/list"}')" "len(d['result']['tools'])>0"
done

check rest-changes "$(curl -sS "$BASE/api/changes?since=2026-06-01&area=uk-rates")" "d['area']=='uk-rates' and 'cite' in d"
check rest-rates-envelope "$(curl -sS "$BASE/api/uk/rates?category=vat")" "'freshness' in d and 'cite' in d"

R="$(curl -sS -D - "$BASE/r/uk/rates?category=income-tax")"
if grep -q 'X-Robots-Tag: noindex' <<<"$R" && grep -q 'income-tax-rates' <<<"$R" && grep -q 'Personal Allowance' <<<"$R"; then echo "PASS /r page"; else echo "FAIL /r page"; FAILED=1; fi
CODE="$(curl -s -o /dev/null -w '%{http_code}' "$BASE/api/cron/watch-sources")"
if [ "$CODE" = "401" ]; then echo "PASS cron unauth"; else echo "FAIL cron unauth ($CODE)"; FAILED=1; fi

[ "$FAILED" = 0 ] && echo "ALL PASS" || { echo "SOME FAILED"; exit 1; }
```
